x-to-nitter-random.user.js 
// ==UserScript==
// @name         X.com to Random Nitter Redirector (5 Instances)
// @namespace    https://github.com/
// @version      1.0
// @description  Automatically redirects any x.com or twitter.com URL to one of 5 Nitter instances randomly!
// @match        *://x.com/*
// @match        *://www.x.com/*
// @match        *://mobile.x.com/*
// @match        *://twitter.com/*
// @match        *://www.twitter.com/*
// @match        *://mobile.twitter.com/*
// @run-at       document-start
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    const instances = [
        "https://nitter.catsarch.com",
        "https://nitter.tiekoetter.com",
        "https://shitter.thepixora.com",
        "https://nitter.jaydenha.uk",
        "https://songlark.jaydenha.uk"
    ];

    // Pick 1 random instance from the 5 URLs
    const randomHost = instances[Math.floor(Math.random() * instances.length)];

    // Preserve path, query parameters, and hash (e.g. /david or /david/status/123)
    const targetUrl = randomHost + window.location.pathname + window.location.search + window.location.hash;

    window.location.replace(targetUrl);
})();
